### Bước 3: Tạo Nginx template cho consul-template

```bash
cat > nginx/nginx.ctmpl << 'EOF'
# NGINX UPSTREAM — Auto-generated by consul-template
# DO NOT EDIT MANUALLY

upstream order_backend {
    server 127.0.0.1:1;  # placeholder

    {{ range service "order-service" "any" }}
    server {{ .Address }}:{{ .Port }} max_fails=3 fail_timeout=30s;
    {{ end }}
}

server {
    listen 80;
    server_name _;

    location /health {
        return 200 'OK';
        add_header Content-Type text/plain;
    }

    location /api/orders/ {
        proxy_pass http://order_backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Upstream-Addr $upstream_addr;
        proxy_connect_timeout 5s;
        proxy_read_timeout 30s;
    }

    access_log /var/log/nginx/access.log;
    error_log  /var/log/nginx/error.log warn;
}
EOF
```

### Bước 4: Tạo consul-template config

```bash
cat > consul-template.hcl << 'EOF'
consul {
  address = "consul:8500"
  retry {
    attempts = 5
    backoff = "250ms"
  }
}

template {
  source      = "/etc/consul-template/nginx.ctmpl"
  destination = "/etc/nginx/conf.d/upstream.conf"
  command     = "sh -c 'nginx -t && nginx -s reload'"
  command_timeout = "30s"

  # Debounce: đợi 3-10s sau thay đổi cuối
  wait {
    min = "3s"
    max = "10s"
  }

  # Random splay 2s để tránh thundering herd
  splay = "2s"

  error_on_missing_key = false
  error_fatal          = true
  perms                = 0644
  create_dest_dirs     = true
}

log_level = "info"
reload_signal = "SIGHUP"
kill_signal   = "SIGINT"
EOF
```

|---|---|---|

## Exercise 4: Kong DNS — Stale TTL Fallback khi Service Down

**Mục tiêu**: Observe Kong behavior khi service bị deregister — stale TTL serving.

### Bước 1: Baseline — 3 replicas đang hoạt động

```bash
echo "=== Baseline: 3 replicas active ==="
for i in $(seq 1 10); do
  curl -s http://localhost:8000/api/v1/orders/ | jq -r '.instance'
done | sort | uniq -c

echo "=== Check Consul service count ==="
curl -s http://localhost:8500/v1/catalog/service/order-service | jq 'length'
```

### Bước 2: Kill order-3 container

```bash
echo "=== Kill order-3 ==="
docker stop order-3

# Consul health check sẽ mark order-3 unhealthy sau ~10s
sleep 15

echo "=== Consul: order-3 status ==="
curl -s http://localhost:8500/v1/health/service/order-service | jq '.[] | {Instance: .Service.ID, Status: .Checks[0].Status}'
```

### Bước 3: Observe Kong behavior (stale TTL)

```bash
echo "=== Kong routing immediately after kill (0-4s, stale window) ==="
for i in $(seq 1 10); do
  STATUS=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:8000/api/v1/orders/)
  INSTANCE=$(curl -s http://localhost:8000/api/v1/orders/ | jq -r '.instance')
  echo "HTTP $STATUS | instance=$INSTANCE"
  sleep 0.5
done

# Expected trong stale TTL (0-4s):
# - order-3 vẫn có thể được chọn (stale cache)
# - Kong gửi request đến dead container → 502 hoặc connection refused

# Sau stale TTL (4s+):
# - Kong query lại DNS
# - order-3 removed from DNS response
# - Kong chỉ gửi đến order-1 và order-2
```

### Bước 4: Measure stale TTL window

```bash
echo "=== Testing stale TTL behavior (30 requests) ==="
echo "Timestamp | HTTP Status | Instance"
for i in $(seq 1 30); do
  TS=$(date +%s)
  STATUS=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:8000/api/v1/orders/)
  INSTANCE=$(curl -s http://localhost:8000/api/v1/orders/ | jq -r '.instance')
  echo "$TS | HTTP $STATUS | $INSTANCE"
  sleep 0.3
```

```bash
_format_version: "3.0"
```
